fail2ban ssh:

install fail2ban:

# apt install fail2ban

configure:

root@test-client:/home/ger# vi /etc/fail2ban/jail.local

[DEFAULT]
bantime  = 1h
findtime = 10m
maxretry = 5
# whitelist your own IPs so you don't lock yourself out
ignoreip = 127.0.0.1/8 ::1 <your-trusted-ip>

[sshd]
enabled = true
port    = ssh          # or your custom port number
backend  = systemd

start/enable:

root@test-client:/home/ger# systemctl enable --now fail2ban
Synchronizing state of fail2ban.service with SysV service script with /usr/lib/systemd/systemd-sysv-install.
Executing: /usr/lib/systemd/systemd-sysv-install enable fail2ban

check:
