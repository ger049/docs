eate a cephfs with erasure-coding pool:

[root@ceph-osd-0 ger]# ceph osd erasure-code-profile ls
default
isa_2_2
isa_4_2

create 'normal' metadata-pool:

[root@ceph-osd-0 ger]# ceph osd pool create rdms-poc-ec_metadata
pool 'rdms-poc-ec_metadata' created

create ec data-pool:

[root@ceph-osd-0 ger]# ceph osd pool create rdms-poc-ec_data erasure isa_2_2
pool 'rdms-poc-ec_data' created

create filesystem:

[root@ceph-osd-0 ger]# ceph fs new rdms-poc-ec rdms-poc-ec_metadata rdms-poc-ec_data
Error EINVAL: pool 'rdms-poc-ec_data' (id '26') is an erasure-coded pool. Use of an EC pool for the default data pool is discouraged; see the online CephFS documentation for more information. Use --force to override.

set allopw-ec-overwrite to true:

[root@ceph-osd-0 ger]# ceph osd pool set rdms-poc-ec_data allow_ec_overwrites true
set pool 26 allow_ec_overwrites to true

create filesystem poging 2:

[root@ceph-osd-0 ger]# ceph fs new rdms-poc-ec rdms-poc-ec_metadata rdms-poc-ec_data --force
  Pool 'rdms-poc-ec_data' (id '26') has pg autoscale mode 'on' but is not marked as bulk.
  Consider setting the flag by running
    # ceph osd pool set rdms-poc-ec_data bulk true
new fs with metadata pool 25 and data pool 26

create keyring for new filesystem:

ceph auth get-or-create client.rdms-poc-ec \
  mds 'allow rw fsname=rdms-poc-ec' \
  osd 'allow rw pool=rdms-poc-ec_data, allow rw pool=rdms-poc-ec_metadata' \
  mon 'allow r fsname=rdms-poc-ec' \
  -o /etc/ceph/ceph.client.rdms-poc-ec.keyring

copy /etc/ceph/ceph.client.rdms-poc-ec.keyring to client
