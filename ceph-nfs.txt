ceph-nfs:

share a cephfs over nfs using the ceph-nfs-server:

root@test-ceph:/home/ger# cephadm shell
Inferring fsid 76623c89-407e-11f1-8b9f-bc2411e1209d
Inferring config /var/lib/ceph/76623c89-407e-11f1-8b9f-bc2411e1209d/mon.test-ceph/config
Using ceph image with id 'af0c5903e901' and tag 'v19' created on 2025-07-17 21:53:27 +0200 CEST
quay.io/ceph/ceph@sha256:af0c5903e901e329adabe219dfc8d0c3efc1f05102a753902f33ee16c26b6cee

root@test-ceph:/# hostname
test-ceph

step 1: enable nfs module in Ceph if not already enabled:

root@test-ceph:/# ceph mgr module enable nfs
module 'nfs' is already enabled

step 2: create ceph-nfs-cluster:

root@test-ceph:/# ceph nfs cluster create ceph-nfs01 test-ceph

step 3: check cephfs available:

root@test-ceph:/# ceph fs status
cephfs01 - 1 clients
========

step 4: create NFS export of cephfs
 
root@test-ceph:/# ceph nfs export create cephfs --cluster-id ceph-nfs01 --pseudo-path /export --fsname cephfs01 --path /
{
  "bind": "/export",
  "cluster": "ceph-nfs01",
  "fs": "cephfs01",
  "mode": "RW",
  "path": "/"
}

step 5: check:

root@test-ceph:/# ceph nfs export ls --cluster-id ceph-nfs01
[
  "/export"
]

step 6: extra check:

root@test-ceph:/# ceph nfs export info --cluster-id ceph-nfs01 --pseudo-path /export

root@test-client:/home/ger# mount -t nfs4 test-ceph:/export /nfs
